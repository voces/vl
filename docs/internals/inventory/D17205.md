    type In = { v: i32 }
    type Out = { w: string, k: i32 }
    const rec: In = { v: 9 }
    function run(f: boolean) {
      let m = null
      const j = m ?? rec
      m = { w: "z", k: 4 }
      return j.v
    }
    print(run(true))
    // vl check: rc 0.  vl run: invalid module inside `run`, "type mismatch: expected (ref $0),
    // found (ref $3); $0 is `Out` … $3 is `In`"

Want:

    9

#### Mechanism

The read of `m` before the write is typed `null` and `m ?? rec` is `In`, but the `??` is lowered
off `m`'s storage, which the later write made `Out | null`, so its result block is typed `Out`.
D17203's fix delivers a `null`-typed name read as the destination's null at a plain destination,
a join arm and a union box; the `??` operand and a binding initialised by the read are other
consumers. `fillCoalHoleNodes` re-asks a `??` over the open hole, but this read is narrowed, so it
is never pending.
