    function inc(a: i32): i32 { a + 1 }
    function apply(g, x, y) { return g(x) }
    print(apply(inc, 3, (n) => n * 2))
    // vl check: rc 1, cannot infer a type for parameter `n`

#### Mechanism

From the ablation only. Nothing in `apply` reaches `y`, so no demand gives the lambda's
parameter a type, and an argument lambda with an untyped parameter is refused; a `const`
hole lambda nobody calls is accepted.
