### D2581 — with a std module imported, a nullable-record function whose result a `??` joins with an inline record is declared at a std struct's heap

**check-clean invalid wasm · clause 1 · found grading [D2565](D2565.md)'s close against the #3173 review grid (`aw__ret__zafter`) · the same on master `216770428` · filed 2026-09-26 against master `216770428`**

Repro:

    import { padLeft } from "std:fmt"
    type A = { r: i32 }
    function mkA(n: i32): A | null { if n == 0 { return null } return { r: n } }
    function rt(ka: i32) { return mkA(ka) ?? { w: 3 } }
    function go(ka: i32) {
      const x = rt(ka)
      if x is A { print(x.r) } else { print(x.w) }
    }
    go(0)
    go(7)
    print(padLeft("a", 3, " "))
    // vl check: rc 0.  vl run: invalid module, inside `mkA`:
    // "type mismatch: expected (ref null $type), found (ref null $type)"

#### Mechanism

A prediction. The disassembly shows `mkA`'s body building and nulling `A`'s own heap while its
functype declares `(result (ref null 0))`, and type 0 is a std struct (`{ (ref $list), i32 }`).
Only the signature moved, and the `??` merge redirect (`anonLeafRetMovedOf` /
`anonLeafSupersedeOf`) is a step the signature takes that the body's seed can skip; with std's
rows ahead of the program's, a row index that is right without std names row 0.
