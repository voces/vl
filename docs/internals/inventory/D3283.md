### D3283 — two record literals whose fields hold `R | i32` and `K | string` joins from two functions build invalid top-level code

**check-clean invalid wasm · clause 1 · open · the same on master `569caa36c` · no type parameter · found by the #3297 round-3 review (`s/two2T.vl`) · filed 2026-09-29**

Repro:

    type R = { f: i32 }
    type K = "a" | "b"
    function g(a: R, c: boolean) {
      const r = if c { 5 } else { a }
      return r
    }
    function g2k(a: K, c: boolean) {
      const r = if c { "s" } else { a }
      return r
    }
    const u: R = { f: 1 }
    const k: K = "b"
    const b = { v: g(u, false) }
    const h = { v: g2k(k, true) }
    if b.v is R { print(b.v.f) } else { print("x") }
    print(h.v)
    // vl check: rc 0.  vl run: invalid module inside the module's top-level code (type mismatch: expected i32, found (ref $type))

Want:

    1
    s

#### Mechanism

Not localised past the ablation. The two literals share the field set `{ v }` at two different
unions, one of which (`K | string`) is a literal-union widening. The top level then reads one
literal's field at the other's rep.

#### Ablation
