### D3793 — a generic over a READONLY list of a declared record keeps the lossy structural pin, so a union-arm or recursive element field emits invalid wasm

**check-clean invalid wasm · the readonly face of D3792 · filed 2026-10-06 against master `a37097b64` (#3407 review)**

Repro:

    import { toString } from "std:fmt"
    type Hit = { tick: i32, kind: "hit" }
    type Miss = { reason: string, kind: "miss" }
    type St = { key: i32, evs: (Hit | Miss)[] }
    type H = { ro: readonly St[] }
    function g<T>(x: readonly T[]): T { return x[1] }
    function mk(k: i32): St { return { key: k, evs: [{ tick: k, kind: "hit" }] } }
    const a: St[] = [mk(1), mk(2), mk(3)]
    const h: H = { ro: a }
    const r = g(h.ro)
    const e = r.evs[0]
    if e is Hit { print(r.key.toString() + ":hit " + e.tick.toString()) } else { print("miss") }

Want:

    2:hit 2

#### Mechanism

D3792's `monoDeclRecListPinName` pins a generic's list argument by the element's declared
record name, but returns `""` for a readonly list (`t.aRead == 1`), so `readonly St[]` keeps the
structural render: `kind: "hit"` becomes `string` and the instance's row is not `St`'s, which
the validator rejects inside `g$m0`. Nothing says why readonly is excluded.

| --- | --- | --- |
| `ro: readonly St[]` → `ro: St[]` | runs `2:hit 2` (after #3407) | **yes** |
| recursive `Node` element instead of the arm list | same invalid wasm | no |
