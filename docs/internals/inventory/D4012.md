### D4012 — a wider record inside a list or a nullable, into a declared record parameter, is refused

**open · loud check reject · clause 2 · found by the #3516 review (w16, w17) · filed 2026-10-11 against master `4d0501a9e`**

Repro:

    type Pos = { x: f64, y: f64 }
    type A = { k: i32, y: f64, x: f64 }
    function mv(p: Pos | null) { if p != null { p.x = p.x + 1.0 } }
    const a: A = { k: 7, y: 3.0, x: 2.0 }
    mv(a)
    print(a.x)
    const n: A | null = a
    mv(n)
    print(a.x)
    // vl check: an object value of shape A flowing into Pos drops the field `k`

Want:

    3
    4

#### Mechanism

D3992 rewrites a parameter only when its annotation is the bare record name. `Pos | null` and
`Pos[]` are declined, so a wider record reached through either keeps the width refusal. The
list spelling (`function sum(ps: Pos[])` handed an `A[]`) is the same row: the element would
have to be the type parameter, `W[]` or `W | null`.

#### Grading list

    type Pos = { x: f64, y: f64 }
    type A = { k: i32, y: f64, x: f64 }
    function sum(ps: Pos[]): f64 { let s = 0.0; for p in ps { s = s + p.x } ; s }
    const as: A[] = [{ k: 1, y: 3.0, x: 2.0 }, { k: 2, y: 1.0, x: 4.0 }]
    print(sum(as))
