### D2812 — a function type returning `v128 | null` keyed no signature, so a call through it had none

**closed 2026-09-27 — the filed repro RUNS and prints `103` · closed by keying a nullable vector as the value-union box in `annUnionSigKind`, as the nullable numeric scalars already were · filed 2026-09-27 against master `2bbc5c412`**

Repro:

    import { toString } from "std:fmt"
    function mkv(n: i64): v128 | null {
      if n < 0 { return null }
      __splat_i32x4__(n as% i32)
    }
    function run(fn: (i64) => v128 | null): i32 {
      const a = fn(3)
      const b = fn(-1)
      let s = 0
      if a != null { s = s + __extract_lane_i32x4__(a, 1) }
      if b == null { s = s + 100 }
      s
    }
    print(toString(run(mkv)))
    // master: emitProgram: bare null needs a struct-typed context (and `function-value call arity has no interned signature` without the call to `run`)

#### Mechanism

`v128 | null` reps as the value-union box since #3187, and the named function's own key
(`cloSigKeyExt`) already said so. The annotation producer did not: `annUnionSigKind` listed
the boxed nullable scalars (`i32`, `i64`, `f64`, `f32`) and not `v128`, so the parameter's
function type keyed "" and the call fell to the all-i32 arity functype. The `(v128 | null)`
PARAMETER ran on master because no arity-1 fallback was needed to reach a box there; the
result position is what needed the key.
