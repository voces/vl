### D11030 — an inline record whose field is a literal union adopts a declared row whose field is a plain `string` or `i32`

**closed · check-clean invalid wasm, clause 1, 12 of 36 cells of a literal-union-field matrix · filed 2026-10-23 against master `ad71da511`**

Repro:

    type K = "x" | "y"
    type Rs = { k: string, n: i32 }
    type Ri = { k: i32, n: i32 }
    const rs: Rs = { k: "q", n: 1 }
    const ri: Ri = { k: 7, n: 2 }
    print(rs.k)
    print(ri.k)
    type Ra = { k: K, n: i32 }
    const ra: Ra = { k: "y", n: 3 }
    function va(r: { k: K, n: i32 }): K { r.k }
    print(va(ra))
    type Rb = { k: "a" | "b", n: i32 }
    const rb: Rb = { k: "b", n: 3 }
    function vb(r: { k: "a" | "b", n: i32 }): string { r.k }
    print(vb(rb))
    type Rz = { k: "z", n: i32 }
    const rz: Rz = { k: "z", n: 3 }
    function vz(r: { k: "z", n: i32 }): string { r.k }
    print(vz(rz))
    // master: vl check rc 0; vl run: the emitted module failed to validate

Want:

    q
    7
    y
    b
    z

#### Mechanism

The shape-row lookup by field set (`shapeRowScanK` on the name path, `jrFieldFits` on the
typed path) let a literal-union field (code 0, the atom) sit at a code-3 row so a discriminant
laid out as a string still found its own row (D1460), and let a lone literal `"z"` (code 0 by
spelling) sit at any code-0 row. Neither asked the row's own field type, so an inline
`{k: K}` took the first declared `{k: string}` row and `{k: "z"}` a declared `{k: i32}` row.
The fix (`rowFieldTakesLit`) refuses a code-0 or code-3 row whose own field type is known and
is neither a string literal nor a literal union.
